# Bloc Recruitment Alerts — 候補スキャンと状態管理
#
#   player.bra_candidate_houses : 現在の候補家リスト（variable list, house群）。
#                                  毎月クリアして作り直すため、家側に状態を持たせる必要はない。
#   player.bra_scan_active      : 月次チェーンが生きているかを示す期限付き変数（months=4）。
#                                  毎月のスキャンで期限を延長し続けるだけの自己修復ロック。
#                                  チェーンが何らかの理由で止まっても4ヶ月で自然に失効し、
#                                  quarterly_playable_pulse側からの再起動を妨げない。

# root = プレイヤー（陣営指導者）。毎月呼ばれる。
bra_scan_recruitment_candidates_effect = {
	clear_variable_list = { name = bra_candidate_houses }

	# 他Bloc所属の家を走査（家単位のBloc APIを利用。全キャラクター走査はしない）
	every_confederation = {
		limit = {
			NOT = { this = root.house.house_confederation }
		}
		every_confederation_member_house = {
			save_temporary_scope_as = target_house
			if = {
				limit = { bra_can_recruit_house_trigger = yes }
				debug_log = "bra_bloc_recruitment: candidate found (other bloc) - [scope:target_house.GetName]"
				root = {
					add_to_variable_list = { name = bra_candidate_houses target = scope:target_house }
				}
			}
		}
	}

	# Bloc未所属の家を走査（プレイヤーの実現割配下に限定。全世界走査はしない）
	top_liege = {
		every_vassal_or_below = {
			limit = {
				is_house_head = yes
				any_held_title = { is_noble_family_title = yes }
				NOT = { exists = house.house_confederation }
			}
			house = { save_temporary_scope_as = target_house }
			if = {
				limit = { bra_can_recruit_house_trigger = yes }
				debug_log = "bra_bloc_recruitment: candidate found (independent) - [scope:target_house.GetName]"
				root = {
					add_to_variable_list = { name = bra_candidate_houses target = scope:target_house }
				}
			}
		}
	}
}

# root = プレイヤーだった character。陣営指導者でなくなったときに状態を破棄する。
bra_bloc_recruitment_clear_state_effect = {
	clear_variable_list = { name = bra_candidate_houses }
	remove_variable = bra_scan_active
}
