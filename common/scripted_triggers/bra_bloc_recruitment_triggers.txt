# Bloc Recruitment Alerts — 勧誘可否判定
#
# CK3本体には「勧誘可能」を表す単一のtriggerが存在しない。
# 条件は common/character_interactions/10_tgp_japan_interactions.txt の
# invite_to_house_bloc_interaction (redirect / is_available / is_shown /
# is_valid_showing_failures_only, 概ね1-190行) に直書きされているため、
# ここでその条件を複製する。本体アップデートで乖離しうるので、
# 追従時は上記ファイルと突き合わせて確認すること。

# root = 判定対象character。陣営（Bloc）の指導者かどうか。
# is_aliveを見ているのは、指導者の死亡時に月次チェーン（events/bra_bloc_recruitment_events.txt）
# を確実に停止させるため（死亡characterに対してイベントが動き続けるのを防ぐ）。
bra_is_bloc_leader_trigger = {
	is_alive = yes
	tgp_uses_house_blocs_trigger = yes
	exists = house.house_confederation
	house.house_confederation.leading_house ?= house
}

# root = 勧誘する側character（陣営指導者を想定）
# scope:target_house = 勧誘対象の家
# invite_to_house_bloc_interaction の is_available / is_valid_showing_failures_only
# を再現。custom_tooltipのラップは外し、中の条件のみを残す。
bra_can_recruit_house_trigger = {
	# --- 招待側（root=プレイヤー）の条件 ---
	# every_confederation_member_house / every_vassal_or_below の内側から呼ばれるため、
	# thisは候補側（house または candidate character）になっている。無条件のまま書くと
	# 誤って候補側に対して評価されてしまうため、必ずroot（プレイヤー）へ明示的に切り替える。
	root = {
		tgp_house_bloc_interaction_valid_trigger = yes # is_ruler / DLC / 政体フラグ / 最高位>=county
		tgp_house_bloc_interaction_valid_showing_failures_trigger = yes # house head + noble family title保有
		is_at_war = no
		house.house_head ?= this
		OR = {
			is_landed = yes
			tgp_is_ceremonial_liege_trigger = yes
		}
		# 陣営を持っているなら、自分がその陣営の指導者家であること
		trigger_if = {
			limit = { exists = house.house_confederation }
			house.house_confederation.leading_house.house_head ?= this
		}
	}

	# --- 対象側（scope:target_house）の条件 ---
	scope:target_house = {
		# 自分の家ではない
		this != root.house
		# 自分の陣営には所属していない（同一陣営は対象外）
		trigger_if = {
			limit = { exists = root.house.house_confederation }
			NOT = { house_confederation ?= root.house.house_confederation }
		}
		house_head ?= {
			is_alive = yes
			tgp_house_bloc_interaction_valid_trigger = yes
			tgp_house_bloc_interaction_valid_showing_failures_trigger = yes
			is_at_war = no
			top_liege = root.top_liege # 同じ実現割に属する

			# 封臣関係の制約（独立君主なら不要）
			trigger_if = {
				limit = { is_independent_ruler = no }
				OR = {
					liege = root.liege # 同じ直上主君
					liege = root # rootが自分の主君
					AND = { # rootが自分の主君の主君で、rootが帝国位未満
						liege.liege = root
						root = { highest_held_title_tier < tier_empire }
					}
					liege = { # 自分の主君の家がrootの陣営に所属
						house.house_confederation = root.house.house_confederation
					}
				}
			}
		}
		# 陣営リーダーは引き抜けない
		NOT = {
			exists = house_confederation
			house_confederation.leading_house.house_head ?= house_head
		}
		# 離脱直後のクールダウン中ではない（自分の陣営に対して）
		trigger_if = {
			limit = { exists = root.house.house_confederation }
			NOT = {
				any_in_list = {
					variable = left_bloc_cooldown
					root.house.house_confederation ?= this
				}
			}
		}
		# 加入直後（引き抜かれたばかり）ではない
		NOT = { has_variable = recently_joined_bloc }
	}
}
# 検知対象は「A: Bloc未所属」「B: 他Bloc所属」の家だが、上記条件はこの2つで
# 網羅的（scope:target_houseがhouse_confederationを持つか持たないかの両方を許容
# している）ため、A/Bを判別する追加条件は不要。

# root = 招待する側character（プレイヤー）、scope:target_house = 勧誘対象の家
# 「送れるか」だけでなく、送った場合に実際に受け入れられそうかまで判定する。
#
# 【この判定はあくまで近似であり、断定的な保証ではない】
# invite_to_house_bloc_interaction の ai_accept はCK3の型定義上 mtth
# （common/character_interactions/_character_interactions.info 参照）で、
# house_bloc_desire_to_join_modifiers / house_relation_ai_accept_modifiers
# （common/scripted_modifiers/10_tgp_japan_modifiers.txt）という専用文法の
# scripted_modifier（modifier= / opinion_modifier=）で組み立てられている。
# この文法はscript_value（save_temporary_scope_value_asのvalue={}）とは
# 別物で両立できないため、本体の計算式をそのまま再現することはできない。
# そのため、ai_acceptの中で特に比重が大きいOpinion（±25分）とHouse
# Relation（feud -50 / rivalry -25分）だけを通常のtriggerとして直接抽出し、
# 「Opinionが極端に悪くなく、House Relationも険悪でない」ことを目安として
# 判定する。相対的な威信・影響力・軍事力による加減点（各最大±X分）は
# 反映されないため、これらが強く効くケースでは実際の受諾可否とずれうる。
bra_will_join_bloc_trigger = {
	bra_can_recruit_house_trigger = yes
	scope:target_house = {
		house_head = {
			opinion = {
				target = root
				value >= 0
			}
			NOT = {
				OR = {
					tgp_has_house_relation_level_trigger = {
						HOUSE_1_MEMBER = this
						HOUSE_2_MEMBER = root
						LEVEL = feud
					}
					tgp_has_house_relation_level_trigger = {
						HOUSE_1_MEMBER = this
						HOUSE_2_MEMBER = root
						LEVEL = rivalry
					}
				}
			}
		}
	}
}
